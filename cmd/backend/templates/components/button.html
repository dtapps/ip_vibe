{{ define "button" }} {{/* 支持的参数： .Type: 按钮类型/颜色，如 "primary", "danger"。默认 "primary" .IconName: 图标名称 .Text: 按钮文本 .URL: 可选，API 接口地址（用于抽象化 DELETE 操作） .OnClick: 可选，自定义的 JavaScript onclick 字符串 (例如: onDelete('123')) .Disabled: 可选，布尔值，是否禁用 .Tooltip: 可选，如果按钮被禁用，此处提供提示文本 (title) .NoReload: 可选，布尔值。如果为 true，则 API 删除成功后不自动刷新页面。 */}} {{/* 设置默认值 */}} {{ $type := .Type }} {{ if not $type }} {{ $type = "primary" }} {{ end }} {{ $iconName := .IconName }} {{ $text := .Text }} {{ $url := .URL }} {{ $onClick := .OnClick }} {{ $disabled := .Disabled }} {{ $tooltip := .Tooltip }} {{ $noReload := .NoReload }} {{ if not $noReload }} {{ $noReload = false }} {{ end }} {{/* 如果禁用且提供了提示文本，则使用 Tooltip 包装器 */}} {{ if and $disabled $tooltip }} {{ end }} {{/* 结束 Tooltip 包装器 */}} {{ if and $disabled $tooltip }} {{ end }} {{ if $url }} {{ end }} {{ if $onClick }} {{ end }} {{ end }}
